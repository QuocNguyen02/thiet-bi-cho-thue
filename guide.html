Hướng dẫn sử dụng — Thiết Bị Cho Thuê

App chạy trên Windows và macOS. Dữ liệu nằm trên một file Google Sheet duy nhất, dùng mãi mãi: các chi nhánh nhập trên app, chi nhánh kia thấy ngay. Lịch sử thuê lưu lại hết, xem lại theo từng tháng.

Tổng quan 1. Tạo Google Sheet 2. Cài app 3. Kết nối 4. Nhập thiết bị Dùng hằng ngày Xem lại tháng trước iPhone Nhiều chi nhánh Câu hỏi thường gặp

  3. Xóa hết chữ có sẵn trong ô soạn mã, rồi bấm nút dưới đây và dán (Ctrl+V / Cmd+V) vào:

/***** ĐIỂM VÀO WEB APP *****/
function doGet(e) {
  return handle_((e && e.parameter) || {});
}

function doPost(e) {
  var body;
  try {
    body = JSON.parse((e && e.postData && e.postData.contents) || '{}');
  } catch (err) {
    return json_({ ok: false, error: 'Dữ liệu gửi lên không hợp lệ' });
  }
  return handle_(body);
}

function handle_(req) {
  try {
    var me = auth_(req.key);
    if (!me) return json_({ ok: false, code: 'AUTH', error: 'Sai mã bảo mật' });
    req.me = me;
    var action = req.action || 'sync';
    var lock = LockService.getScriptLock();
    lock.waitLock(25000);
    try {
      var result;
      if (action === 'ping') {
        Object.keys(TABLES).forEach(ensure_);
        var ss = SpreadsheetApp.getActiveSpreadsheet();
        result = {
          ok: true, name: ss.getName(), version: VERSION, serverTime: now_(),
          branch: me.branch, owner: me.owner, branches: branchNames_(),
          url: me.owner ? ss.getUrl() : '',   // link file Sheet (chứa doanh thu mọi chi nhánh) chỉ gửi cho chủ
        };
      } else if (action === 'sync') result = sync_(req);
      else if (action === 'saveRental') result = saveRental_(req);
      else if (['add', 'update', 'delete'].indexOf(action) >= 0) {
        if (!TABLES[req.table]) result = { ok: false, error: 'Bảng không hợp lệ: ' + req.table };
        else if (req.table === 'rentals' && action === 'add') result = addRental_(clean_('rentals', req.item || {}), req);
        else if (req.table === 'devices' && action === 'add' && Number(req.qty) > 1) result = addDevices_(req.item || {}, Number(req.qty), req);
        else if (action === 'add') result = add_(req.table, req.item || {}, req);
        else if (action === 'update') result = update_(req.table, String((req.item && req.item.id) || req.id || ''), req.item || {}, req);
        else result = remove_(req.table, String(req.id || ''), req);
      } else result = { ok: false, error: 'Hành động không hỗ trợ: ' + action };
      SpreadsheetApp.flush();
      return json_(result);
    } finally {
      lock.releaseLock();
    }
  } catch (err) {
    return json_({ ok: false, error: String((err && err.message) || err) });
  }
}

/***** PHÂN QUYỀN CHI NHÁNH *****/
// → { branch: 'Chi nhánh 1', owner: false } | { branch: '', owner: true } | null (sai mã)
function auth_(key) {
  key = String(key || '');
  if (!key) return null;
  if (OWNER_KEY && key === OWNER_KEY) return { branch: '', owner: true };
  if (Object.prototype.hasOwnProperty.call(BRANCH_KEYS, key)) return { branch: String(BRANCH_KEYS[key]).trim(), owner: false };
  return null;
}

function sameBranch_(a, b) {
  return String(a || '').trim().toLowerCase() === String(b || '').trim().toLowerCase();
}

// Người đang gọi có được xem đầy đủ / sửa đơn này không:
// chủ → mọi đơn; chi nhánh → đơn của chi nhánh mình, hoặc đơn mình đã tạo hộ chi nhánh khác
function ownsRental_(req, rental) {
  return req.me.owner || sameBranch_(rental.branch, req.me.branch) || sameBranch_(rental.createdBranch, req.me.branch);
}

// Danh sách tên chi nhánh (theo BRANCH_KEYS, không trùng)
function branchNames_() {
  var out = [];
  Object.keys(BRANCH_KEYS).forEach(function (k) {
    var b = String(BRANCH_KEYS[k]).trim();
    if (b && !out.some(function (x) { return sameBranch_(x, b); })) out.push(b);
  });
  return out;
}

// Tên chi nhánh chuẩn (đúng chữ hoa/thường như trong BRANCH_KEYS), '' nếu không có
function canonBranch_(b) {
  return branchNames_().filter(function (x) { return sameBranch_(x, b); })[0] || '';
}

// Đơn mới: chi nhánh nhận đơn = chi nhánh được chọn (mặc định là chi nhánh mình);
// chi nhánh tạo đơn = chi nhánh của mã đang dùng (chủ: để trống). Trả về lỗi hoặc null.
function prepareNewRental_(item, req) {
  var want = String(item.branch || '').trim() || req.me.branch;
  var b = canonBranch_(want);
  if (!b) return { ok: false, error: want ? 'Chi nhánh "' + want + '" không có trong BRANCH_KEYS' : 'Chưa chọn chi nhánh cho đơn thuê' };
  item.branch = b;
  item.createdBranch = req.me.owner ? '' : req.me.branch;
  return null;
}

// Đơn của chi nhánh khác: chỉ giữ thông tin để biết thiết bị bận lúc nào.
// Bỏ toàn bộ tiền, giá, ghi chú và mọi thông tin khách hàng.
function maskRental_(r) {
  var items = [];
  try {
    items = JSON.parse(r.itemsJson || '[]').map(function (i) { return { id: i.id, name: i.name, serial: i.serial || '' }; });
  } catch (err) { items = []; }
  return {
    id: r.id, status: r.status, branch: r.branch,
    items: r.items, itemsJson: JSON.stringify(items),
    rentDate: r.rentDate, returnDate: r.returnDate, returnedDate: r.returnedDate,
    updatedAt: r.updatedAt, masked: true,
  };
}

function forbidden_(what) {
  return { ok: false, code: 'FORBIDDEN', error: (what || 'Đơn này') + ' thuộc chi nhánh khác — chỉ chi nhánh đó mới sửa được' };
}

// Tạo đơn: chi nhánh bắt buộc và phải là một chi nhánh có thật trong BRANCH_KEYS
function addRental_(item, req) {
  var err = prepareNewRental_(item, req);
  return err || add_('rentals', item, req);
}

// Thêm nhiều thiết bị giống nhau một lần (VD 2 máy cùng loại chưa có serial): mỗi máy là 1 dòng riêng,
// serial để trống — bổ sung sau bằng cách sửa từng máy.
function addDevices_(src, qty, req) {
  qty = Math.max(1, Math.min(50, Math.floor(qty) || 1));
  var items = [];
  for (var i = 0; i < qty; i++) {
    var item = {};
    Object.keys(src).forEach(function (k) { item[k] = src[k]; });
    if (qty > 1) item.serial = '';   // nhiều máy: serial không thể giống nhau → để trống, điền sau
    var res = add_('devices', item, req);
    if (!res.ok) return res;
    items.push(res.item);
  }
  return { ok: true, item: items[0], items: items };
}

/***** ĐỒNG BỘ *****/
// since = thời điểm lần đồng bộ trước → chỉ gửi các dòng đã đổi + danh sách ID (để app biết dòng nào bị xóa)
function sync_(req) {
  var serverTime = now_();
  var since = String(req.since || '');
  var tables = {};
  Object.keys(TABLES).forEach(function (name) { tables[name] = readTable_(name, since); });   // khách hàng, thiết bị: dùng chung
  if (!req.me.owner) {
    tables.rentals.rows = tables.rentals.rows.map(function (r) { return ownsRental_(req, r) ? r : maskRental_(r); });
  }
  return {
    ok: true, serverTime: serverTime, full: !since, tables: tables,
    branch: req.me.branch, owner: req.me.owner, branches: branchNames_(),
  };
}

function readTable_(name, since) {
  var info = ensure_(name), sh = info.sh, last = sh.getLastRow();
  if (last < 2) return { rows: [], ids: [] };
  var values = sh.getRange(2, 1, last - 1, info.width).getValues();
  var rows = [], ids = [], stamp = now_();
  for (var i = 0; i < values.length; i++) {
    if (isBlankRow_(values[i])) continue;
    var it = toItem_(info, values[i]);
    if (!it.id || !it.updatedAt) {               // dòng gõ tay trên Sheet: cấp ID + mốc thời gian
      if (!it.id) {
        it.id = newId_(info.t.prefix);
        sh.getRange(i + 2, info.cols.id).setValue(it.id);
      }
      if (!it.updatedAt) {
        it.updatedAt = stamp;
        sh.getRange(i + 2, info.cols.updatedAt).setValue(stamp);
      }
    }
    ids.push(it.id);
    if (!since || it.updatedAt >= since) rows.push(it);
  }
  return { rows: rows, ids: ids };
}

/***** THÊM / SỬA / XÓA *****/
function add_(name, src, req) {
  var info = ensure_(name), t = info.t;
  var item = clean_(name, src);
  for (var i = 0; i < t.required.length; i++) {
    if (!item[t.required[i]]) return { ok: false, error: 'Thiếu ' + labelOf_(name, t.required[i]) };
  }
  var stamp = now_();
  item.id = name === 'rentals' ? nextRentalId_(info) : newId_(t.prefix);
  if (info.cols.createdAt) item.createdAt = stamp;
  if (info.cols.createdBy) item.createdBy = by_(req);
  item.updatedAt = stamp;
  item.updatedBy = by_(req);
  Object.keys(t.enums).forEach(function (k) { item[k] = item[k] || t.enums[k][0]; });
  writeRow_(info, info.sh.getLastRow() + 1, item);
  log_(t.label, 'Thêm', item, req, '');
  return { ok: true, item: item };
}

function update_(name, id, src, req) {
  var info = ensure_(name), t = info.t;
  var rowNum = findRow_(info, id);
  if (!rowNum) return { ok: false, code: 'NOT_FOUND', error: t.label + ' không còn tồn tại (có thể đã bị xóa)' };
  var current = toItem_(info, info.sh.getRange(rowNum, 1, 1, info.width).getValues()[0]);
  if (name === 'rentals' && !ownsRental_(req, current)) return forbidden_();
  if (!req.force && req.expectedUpdatedAt && current.updatedAt && current.updatedAt !== req.expectedUpdatedAt) {
    return { ok: false, code: 'CONFLICT', error: t.label + ' vừa được người khác cập nhật', current: current };
  }
  var changes = clean_(name, src);
  if (name === 'rentals') delete changes.createdBranch;   // chi nhánh tạo đơn do máy chủ ghi, không sửa được
  // đổi chi nhánh nhận đơn: phải là chi nhánh có thật
  if (name === 'rentals' && 'branch' in changes) {
    var b = canonBranch_(changes.branch);
    if (!b) return { ok: false, error: 'Chi nhánh "' + changes.branch + '" không có trong BRANCH_KEYS' };
    changes.branch = b;
  }
  var next = {};
  Object.keys(current).forEach(function (k) { next[k] = current[k]; });
  Object.keys(changes).forEach(function (k) { next[k] = changes[k]; });
  for (var i = 0; i < t.required.length; i++) {
    if (!next[t.required[i]]) return { ok: false, error: 'Thiếu ' + labelOf_(name, t.required[i]) };
  }
  next.id = current.id;
  next.updatedAt = now_();
  next.updatedBy = by_(req);
  writeRow_(info, rowNum, next);
  log_(t.label, 'Sửa', next, req, diff_(name, current, next));
  return { ok: true, item: next };
}

function remove_(name, id, req) {
  var info = ensure_(name);
  var rowNum = findRow_(info, id);
  if (!rowNum) return { ok: true, deleted: false };
  var current = toItem_(info, info.sh.getRange(rowNum, 1, 1, info.width).getValues()[0]);
  if (name === 'rentals' && !ownsRental_(req, current)) return forbidden_();
  if (!req.force && req.expectedUpdatedAt && current.updatedAt && current.updatedAt !== req.expectedUpdatedAt) {
    return { ok: false, code: 'CONFLICT', error: info.t.label + ' vừa được người khác cập nhật', current: current };
  }
  info.sh.deleteRow(rowNum);
  log_(info.t.label, 'Xóa', current, req, '');
  return { ok: true, deleted: true };
}

/**
 * Lưu đơn thuê cùng thông tin khách:
 *  - customer.id có sẵn → cập nhật thông tin khách (chỉ các ô có nhập)
 *  - không có id nhưng trùng SĐT với khách cũ → dùng khách cũ
 *  - còn lại → tạo khách mới
 */
function saveRental_(req) {
  var rental = req.rental || {};
  var changes = clean_('rentals', rental);
  // kiểm tra quyền + chi nhánh TRƯỚC khi đụng tới khách hàng
  if (rental.id) {
    var rInfo = ensure_('rentals'), rRow = findRow_(rInfo, String(rental.id));
    if (rRow && !ownsRental_(req, toItem_(rInfo, rInfo.sh.getRange(rRow, 1, 1, rInfo.width).getValues()[0]))) return forbidden_();
  } else {
    var err = prepareNewRental_(changes, req);
    if (err) return err;
  }
  if (rental.id && 'branch' in changes) {          // sửa đơn, đổi chi nhánh nhận đơn
    var b = canonBranch_(changes.branch);
    if (!b) return { ok: false, error: changes.branch ? 'Chi nhánh "' + changes.branch + '" không có trong BRANCH_KEYS' : 'Chưa chọn chi nhánh cho đơn thuê' };
    changes.branch = b;
  }
  var customer = null;
  if (req.customer && String(req.customer.name || '').trim()) {
    customer = upsertCustomer_(req.customer, req);
    changes.customerId = customer.id;
    changes.customerName = customer.name;
    changes.phone = customer.phone;
  }
  var res = rental.id ? update_('rentals', String(rental.id), changes, req) : add_('rentals', changes, req);
  if (!res.ok) {
    res.customer = customer;
    return res;
  }
  return { ok: true, rental: res.item, customer: customer };
}

// Khách hàng dùng chung giữa các chi nhánh: trùng SĐT với khách cũ (ở bất kỳ chi nhánh nào) → dùng khách cũ
function upsertCustomer_(src, req) {
  var info = ensure_('customers');
  var c = clean_('customers', src);
  var id = String(src.id || '').trim();
  var byPhone = false;
  var rowNum = id ? findRow_(info, id) : 0;
  var current = rowNum ? toItem_(info, info.sh.getRange(rowNum, 1, 1, info.width).getValues()[0]) : null;
  if (!rowNum && c.phone) {
    var found = findBy_(info, function (it) { return digits_(it.phone) && digits_(it.phone) === digits_(c.phone); });
    if (found) { rowNum = findRow_(info, found.id); current = found; byPhone = true; }
  }
  if (!rowNum) return add_('customers', c, req).item;
  var changed = false, next = {};
  Object.keys(current).forEach(function (k) { next[k] = current[k]; });
  Object.keys(c).forEach(function (k) {
    if (c[k] === '' || String(c[k]) === String(current[k])) return;
    if (byPhone && current[k] !== '') return;   // nhận ra qua SĐT: chỉ điền ô còn trống
    next[k] = c[k];
    changed = true;
  });
  if (!changed) return current;
  next.updatedAt = now_();
  next.updatedBy = by_(req);
  writeRow_(info, rowNum, next);
  log_('Khách hàng', 'Sửa', next, req, diff_('customers', current, next));
  return next;
}

var INFO_CACHE_ = {};
function ensure_(name) {
  if (INFO_CACHE_[name]) return INFO_CACHE_[name];
  var t = TABLES[name];
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(t.sheet) || ss.insertSheet(t.sheet);
  var lastCol = Math.max(sh.getLastColumn(), 1);
  var header = sh.getRange(1, 1, 1, lastCol).getValues()[0].map(function (h) { return String(h).trim(); });
  while (header.length && header[header.length - 1] === '') header.pop();
  var cols = {}, added = false;
  t.fields.forEach(function (f) {
    var i = header.indexOf(f[1]);
    if (i < 0) {
      i = header.length;
      header.push(f[1]);
      sh.getRange(1, i + 1).setValue(f[1]);
      formatColumn_(sh, i + 1, f, t);
      added = true;
    }
    cols[f[0]] = i + 1;
  });
  if (added) styleHeader_(sh, header.length);
  INFO_CACHE_[name] = { sh: sh, t: t, cols: cols, width: header.length };
  return INFO_CACHE_[name];
}

function formatColumn_(sh, col, f, t) {
  var rows = sh.getMaxRows() - 1;
  if (rows > 0) {
    var range = sh.getRange(2, col, rows, 1);
    range.setNumberFormat(f[2]);
    if (t.enums[f[0]]) {
      range.setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(t.enums[f[0]], true).setAllowInvalid(true).build());
    }
  }
  sh.setColumnWidth(col, f[3]);
}

function styleHeader_(sh, width) {
  sh.getRange(1, 1, 1, width).setFontWeight('bold').setBackground('#1F3A5F').setFontColor('#FFFFFF');
  sh.setFrozenRows(1);
}

function writeRow_(info, rowNum, item) {
  var sh = info.sh;
  if (rowNum > sh.getMaxRows()) sh.insertRowsAfter(sh.getMaxRows(), 200);
  var range = sh.getRange(rowNum, 1, 1, info.width);
  var vals = range.getValues()[0];
  var fmts = range.getNumberFormats()[0];
  info.t.fields.forEach(function (f) {
    var c = info.cols[f[0]] - 1;
    var v = item[f[0]];
    vals[c] = v === undefined || v === null ? '' : v;
    fmts[c] = f[2];
  });
  range.setNumberFormats([fmts]);
  range.setValues([vals]);
}

function findRow_(info, id) {
  if (!id) return 0;
  var last = info.sh.getLastRow();
  if (last < 2) return 0;
  var ids = info.sh.getRange(2, info.cols.id, last - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) if (String(ids[i][0]).trim() === id) return i + 2;
  return 0;
}

function findBy_(info, fn) {
  var last = info.sh.getLastRow();
  if (last < 2) return null;
  var values = info.sh.getRange(2, 1, last - 1, info.width).getValues();
  for (var i = 0; i < values.length; i++) {
    if (isBlankRow_(values[i])) continue;
    var it = toItem_(info, values[i]);
    if (fn(it)) return it;
  }
  return null;
}

function nextRentalId_(info) {
  var prefix = 'DT' + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyMMdd') + '-';
  var max = 0, last = info.sh.getLastRow();
  if (last >= 2) {
    info.sh.getRange(2, info.cols.id, last - 1, 1).getValues().forEach(function (r) {
      var s = String(r[0]);
      if (s.indexOf(prefix) === 0) max = Math.max(max, Number(s.slice(prefix.length)) || 0);
    });
  }
  return prefix + ('00' + (max + 1)).slice(-3);
}

function toItem_(info, row) {
  var item = {}, t = info.t;
  t.fields.forEach(function (f) {
    var k = f[0], v = row[info.cols[k] - 1];
    if (v instanceof Date) {
      var tz = Session.getScriptTimeZone();
      var fmt = 'yyyy-MM-dd';
      if (DATETIME_KEYS.indexOf(k) >= 0) fmt = 'yyyy-MM-dd HH:mm:ss';
      else if (RENT_TIME_KEYS.indexOf(k) >= 0 && Utilities.formatDate(v, tz, 'HH:mm') !== '00:00') fmt = 'yyyy-MM-dd HH:mm';
      v = Utilities.formatDate(v, tz, fmt);
    }
    if (t.numeric.indexOf(k) >= 0) item[k] = toNum_(v);
    else item[k] = String(v === null || v === undefined ? '' : v).trim();
  });
  return item;
}

function clean_(name, src) {
  var t = TABLES[name], out = {};
  t.fields.forEach(function (f) {
    var k = f[0];
    if (META.indexOf(k) >= 0 || !(k in src)) return;
    var v = src[k];
    if (t.numeric.indexOf(k) >= 0) out[k] = toNum_(v);
    else out[k] = String(v === null || v === undefined ? '' : v).trim().slice(0, k === 'itemsJson' ? 45000 : 2000);
  });
  Object.keys(t.enums).forEach(function (k) {
    if (k in out && t.enums[k].indexOf(out[k]) < 0) out[k] = t.enums[k][0];
  });
  return out;
}

function toNum_(v) {
  if (typeof v === 'number') return v;
  var s = String(v === null || v === undefined ? '' : v).trim();
  if (!s) return '';
  var n = Number(s.replace(/[^\d-]/g, ''));
  return isNaN(n) ? '' : n;
}

function labelOf_(name, key) {
  var f = TABLES[name].fields.filter(function (x) { return x[0] === key; })[0];
  return f ? f[1] : key;
}

function diff_(name, a, b) {
  var out = [];
  TABLES[name].fields.forEach(function (f) {
    var k = f[0];
    if (META.indexOf(k) >= 0 || k === 'itemsJson') return;
    if (String(a[k]) !== String(b[k])) out.push(f[1] + ': "' + a[k] + '" → "' + b[k] + '"');
  });
  return out.join('; ');
}

function log_(table, action, item, req, detail) {
  try {
    var sh = logSheet_();
    var title = item.name || [item.customerName, item.items].filter(String).join(' — ');
    sh.appendRow([now_(), by_(req), table, action, item.id, title, detail]);
    if (sh.getLastRow() > LOG_MAX_ROWS) sh.deleteRows(2, 1000);
  } catch (err) { /* không chặn thao tác chính nếu ghi nhật ký lỗi */ }
}

function isBlankRow_(row) {
  for (var i = 0; i < row.length; i++) if (String(row[i]).trim() !== '') return false;
  return true;
}

function digits_(s) {
  return String(s || '').replace(/\D/g, '');
}

var ID_SEQ_ = 0;   // tạo nhiều dòng trong cùng 1 mili-giây vẫn không trùng ID
function newId_(prefix) {
  return prefix + Date.now().toString(36).toUpperCase() + (ID_SEQ_++).toString(36).toUpperCase() + Math.floor(Math.random() * 1296).toString(36).toUpperCase();
}

function now_() {
  return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
}

function by_(req) {
  return String(req.by || 'Không rõ').slice(0, 100);
}

}

// Tự cập nhật "Cập nhật lúc" để app nhận ra thay đổi và tránh ghi đè.
function onEdit(e) {
  if (!e || !e.range) return;
  var sheetName = e.range.getSheet().getName();
  var name = Object.keys(TABLES).filter(function (n) { return TABLES[n].sheet === sheetName; })[0];
  if (!name) return;
  var info = ensure_(name);
  var metaCols = META.map(function (k) { return info.cols[k]; }).filter(Boolean);
  var onlyMeta = true;
  for (var c = e.range.getColumn(); c <= e.range.getLastColumn(); c++) {
    if (metaCols.indexOf(c) < 0) { onlyMeta = false; break; }
  }
  if (onlyMeta) return;
  var start = Math.max(e.range.getRow(), 2), end = e.range.getLastRow();
  if (end < start) return;
  var rows = info.sh.getRange(start, 1, end - start + 1, info.width).getValues();
  var stamp = now_();
  rows.forEach(function (r, i) {
    if (isBlankRow_(r)) return;
    var rowNum = start + i;
    if (!String(r[info.cols.id - 1]).trim()) info.sh.getRange(rowNum, info.cols.id).setValue(newId_(info.t.prefix));
    info.sh.getRange(rowNum, info.cols.updatedAt).setValue(stamp);
    info.sh.getRange(rowNum, info.cols.updatedBy).setValue('Sửa trên Google Sheet');
  });
}

  • Mỗi iPhone nhớ cài đặt riêng. Xóa icon khỏi màn hình chính thì phải nhập lại link và mã.

Nhiều chi nhánh cùng dùng

  • Doanh thu tách riêng: mỗi chi nhánh chỉ thấy đơn thuê, doanh thu, tiền đã thu / còn phải thu của chi nhánh mình — kể cả khi xuất Excel. Máy chủ lọc dữ liệu theo mã bảo mật, nên chi nhánh khác không tải được về.
  • Thiết bị dùng chung: mọi chi nhánh cùng xem, thêm, sửa danh mục thiết bị (thiết bị không gắn chi nhánh) và cùng thấy số máy đang cho thuê / đã đặt / còn trống. Máy đang ở chi nhánh khác hiện tên chi nhánh và ngày giờ trả — không thấy tiền của đơn đó.
  • Khách hàng dùng chung: danh sách khách (tên, SĐT, CCCD, địa chỉ) là một danh sách chung cho mọi chi nhánh; khách đã thuê ở chi nhánh này sang chi nhánh kia gõ SĐT là ra. Cột "Số lần thuê / Tổng chi" chỉ tính các đơn của chi nhánh đang xem.
  • Đơn thuê luôn có chi nhánh nhận đơn (mặc định là chi nhánh mình). Chi nhánh này có thể tạo đơn hộ chi nhánh khác: chọn chi nhánh khác ở ô Chi nhánh → doanh thu tính cho chi nhánh được chọn, đơn ghi "tạo bởi …". Chi nhánh tạo hộ vẫn xem và xử lý được đơn đó (giao máy, nhận trả) nhưng không cộng vào doanh thu của mình.
  • Thời gian thuê có giờ: nhập giờ cạnh ngày (gõ 930 hoặc 9h30 → 09:30). Có đủ giờ nhận và giờ trả thì số ngày tính tiền = mỗi 24 giờ một ngày, lẻ làm tròn lên (sửa tay được).
  • Mã chủ thấy doanh thu từng chi nhánh cạnh nhau (không cộng tổng); bấm vào một chi nhánh để chỉ xem chi nhánh đó.
  • App tự lấy dữ liệu mới mỗi 20 giây (chỉnh được trong Cài đặt), và khi bạn quay lại cửa sổ app.
  • Hai bên sửa khác ô của cùng một đơn → app tự gộp. Sửa cùng ô → app hiện giá trị hai bên và hỏi có ghi đè không.
  • Khi tạo đơn, thiết bị đang bị chi nhánh kia cho thuê/đặt trùng ngày sẽ được cảnh báo.
  • Mất mạng: app vẫn xem được dữ liệu lần cuối, nhưng chưa lưu được cho đến khi có mạng.
  • Có thể xem/sửa trực tiếp trên Google Sheet — đừng đổi tên các tab và đừng sửa cột "Chi tiết (không sửa tay)".

Câu hỏi thường gặp

Bấm "Đi tới … (không an toàn)" xong không hiện link / mở ra Mail / cửa sổ trắng / báo lỗi

  1. Lấy lại link: Apps Script → Triển khai → Quản lý triển khai → sao chép URL ở mục Ứng dụng web.
  2. Mở ra ứng dụng Mail: do bấm nhầm vào địa chỉ email trên trang của Google. Đóng Mail, quay lại trang Google, bấm Nâng cao → Đi tới … (không an toàn), rồi cuộn xuống bấm Tiếp tục / Cho phép.
  3. Đăng nhập nhiều tài khoản Google (nguyên nhân hay gặp nhất): cửa sổ cấp quyền bị lẫn tài khoản. Mở cửa sổ ẩn danh của Chrome (Cmd + Shift + N), chỉ đăng nhập tài khoản chủ file Sheet, mở lại Sheet → Apps Script và làm lại bước Cấp quyền trước.
  4. Dùng Safari: Safari hay chặn cửa sổ bật lên của Google. Nên dùng Google Chrome; hoặc trong Safari vào Cài đặt → Trang web → Cửa sổ bật lên, chọn Cho phép cho script.google.com.
  5. Tài khoản Google công ty/trường học có thể bị quản trị viên chặn cấp quyền cho mã chưa xác minh → dùng tài khoản Gmail cá nhân để tạo file Sheet.

Thêm chi nhánh mới (VD: Vĩnh Long)

  1. Mở Apps Script, thêm một dòng vào BRANCH_KEYS: 'VL@2026-q8w': 'Vĩnh Long',
  2. Lưu → Triển khai → Quản lý triển khai → ✏️ → Phiên bản mới → Triển khai (link /exec giữ nguyên).
  3. Cài app ở máy Vĩnh Long, nhập link /exec và mã của Vĩnh Long.

Vĩnh Long thấy ngay toàn bộ thiết bị dùng chung và máy nào đang bận; chỉ thấy đơn, khách, doanh thu của Vĩnh Long. Các chi nhánh khác không cần làm gì — danh sách chi nhánh trong app tự cập nhật.

Báo "Google yêu cầu đăng nhập"

Ở bước 1 chưa chọn Người có quyền truy cập: Bất kỳ ai. Vào Apps Script → Triển khai → Quản lý triển khai → ✏️ sửa lại.

Báo "Sai mã bảo mật"

Mã trong app không khớp mã nào trong BRANCH_KEYS / OWNER_KEY (phân biệt chữ hoa/thường). Nhớ triển khai phiên bản mới sau khi sửa mã.

Dán nhầm link file Google Sheet

App sẽ nhắc. Cần link Web App kết thúc bằng /exec (bước 1, mục 8).

Cập nhật mã Apps Script khi có bản mới

Dán mã mới đè lên → Lưu → Triển khai → Quản lý triển khai → ✏️ → Phiên bản: Phiên bản mới → Triển khai. Link /exec giữ nguyên, app không cần cài lại. Dữ liệu cũ giữ nguyên.

Bảo mật

Ai có link Web App và mã của một chi nhánh thì đọc/ghi được dữ liệu của chi nhánh đó (cùng thiết bị, khách hàng dùng chung). Mỗi chi nhánh chỉ giữ mã của mình; mã chủ chỉ chủ giữ. Muốn đổi mã: sửa BRANCH_KEYS / OWNER_KEY, triển khai phiên bản mới, rồi đổi mã trong app ở các máy.

Sao lưu

Google Sheet tự lưu lịch sử phiên bản (Tệp → Nhật ký phiên bản). Muốn chắc hơn: thỉnh thoảng Tệp → Tạo bản sao.
